function getProgramSheetHeadersMap(progSheet) {
  const lastCol = Math.max(progSheet.getLastColumn(), 13);
  const headers = progSheet.getRange(1, 1, 1, lastCol).getValues()[0] || [];
  let colCoord = headers.indexOf('Điều Hành Tour');
  let colCoordPhone = headers.indexOf('SDT Điều Hành');
  let colLogistics = headers.indexOf('Phụ Trách Hậu Cần');
  let colLogisticsPhone = headers.indexOf('SDT Hậu Cần');
  let colDriveId = headers.indexOf('Folder Drive ID');
  let colDriveLink = headers.indexOf('Folder Drive Link');
  let colEndDate = headers.indexOf('Ngày Kết Thúc');

  if (colCoordPhone === -1 || colLogisticsPhone === -1 || colEndDate === -1) {
    progSheet.getRange(1, 1, 1, 13).setValues([[
      'Mã Chương Trình', 'Tên Chương Trình', 'Ngày Thực Hiện',
      'Trạng Thái', 'Thông Báo Admin', 'Ngày Tạo',
      'Điều Hành Tour', 'SDT Điều Hành', 'Phụ Trách Hậu Cần', 'SDT Hậu Cần',
      'Folder Drive ID', 'Folder Drive Link', 'Ngày Kết Thúc'
    ]]);
    progSheet.getRange('A1:M1').setFontWeight('bold').setBackground('#d9ead3');
    colCoord = 6;
    colCoordPhone = 7;
    colLogistics = 8;
    colLogisticsPhone = 9;
    colDriveId = 10;
    colDriveLink = 11;
    colEndDate = 12;
  }

  return {
    colCoord: colCoord !== -1 ? colCoord : 6,
    colCoordPhone: colCoordPhone !== -1 ? colCoordPhone : 7,
    colLogistics: colLogistics !== -1 ? colLogistics : 8,
    colLogisticsPhone: colLogisticsPhone !== -1 ? colLogisticsPhone : 9,
    colDriveId: colDriveId !== -1 ? colDriveId : 10,
    colDriveLink: colDriveLink !== -1 ? colDriveLink : 11,
    colEndDate: colEndDate !== -1 ? colEndDate : 12
  };
}

function normalizeVehicleKey(str) {
  return String(str || '').trim().replace(/\s+/g, ' ').toLowerCase();
}

function removeVietnameseTones(str) {
  let s = String(str || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  s = s.replace(/[đĐ]/g, m => m === 'đ' ? 'd' : 'D');
  return s.trim();
}

function doGet(e) {
  try {
    if (e && e.parameter && e.parameter.action === 'getDriveImage') {
      const fileId = e.parameter.fileId;
      if (!fileId) return ContentService.createTextOutput('Missing fileId');
      const file = DriveApp.getFileById(fileId);
      const blob = file.getBlob();
      return ContentService.createTextOutput(Utilities.base64Encode(blob.getBytes()))
        .setMimeType(ContentService.MimeType.TEXT);
    }
  } catch (err) {
    return ContentService.createTextOutput('Lỗi đọc ảnh: ' + err.toString());
  }
  return handleRequest(e);
}

function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  try {
    const ss = initSheets();
    let params = {};
    if (e.postData && e.postData.contents) {
      params = JSON.parse(e.postData.contents);
    } else if (e.parameter) {
      params = e.parameter;
    }

    // Đón Webhook Zalo hoặc Revoke Consent
    if (params.event_name === 'user_revoke_consent' || !params.action) {
      return jsonResponse({ error: 0, message: 'Đã tiếp nhận yêu cầu xóa dữ liệu' });
    }

    const action = params.action || 'getInitData';

    // Các tác vụ đọc dữ liệu nhanh (không cần chờ Lock)
    if (action === 'getInitData') return jsonResponse(getInitData(ss, params));
    if (action === 'getVehicleData') return jsonResponse(getVehicleData(ss, params));
    if (action === 'getSummaryData') return jsonResponse(getSummaryData(ss, params));
    if (action === 'getTourDetails') return jsonResponse(getTourDetails(ss, params));
    if (action === 'getClassStudentList') return jsonResponse(getClassStudentList(ss, params));
    if (action === 'getAttendanceRecord') return jsonResponse(getAttendanceRecord(ss, params));
    if (action === 'getGeminiApiKeyStatus') return jsonResponse(getGeminiApiKeyStatus(ss, params));

    // Riêng tác vụ ping GPS tự động 15s: dùng lock ngắn 5s để tránh nghẽn hàng đợi
    if (action === 'updateGpsAuto15s') {
      const pingLock = LockService.getScriptLock();
      if (pingLock.tryLock(5000)) {
        try {
          const res = updateStatusAndGps(ss, params, true);
          SpreadsheetApp.flush();
          return jsonResponse(res);
        } finally {
          pingLock.releaseLock();
        }
      } else {
        return jsonResponse({ success: false, message: 'Server busy, skip tick' });
      }
    }

    // Các tác vụ ghi dữ liệu tiêu chuẩn (Khóa 30 giây)
    const lock = LockService.getScriptLock();
    lock.waitLock(30000);
    try {
      let result = {};
      if (action === 'saveVehicleData') {
        result = saveVehicleData(ss, params);
      } else if (action === 'updateStatusOnly') {
        result = updateStatusAndGps(ss, params, false);
      } else if (action === 'createProgram') {
        result = createProgram(ss, params);
      } else if (action === 'updateProgramStatus') {
        result = updateProgramStatus(ss, params);
      } else if (action === 'updateProgramNotice') {
        result = updateProgramNotice(ss, params);
      } else if (action === 'updateTourDetails') {
        result = updateTourDetails(ss, params);
      } else if (action === 'saveTourClassLists') {
        result = saveTourClassLists(ss, params);
      } else if (action === 'uploadAndAnalyzeAttendanceImage') {
        result = uploadAndAnalyzeAttendanceImage(ss, params);
      } else if (action === 'saveAttendanceRecord') {
        result = handleSaveAttendanceRecord(ss, params);
      } else if (action === 'createTourDriveFolder') {
        result = handleCreateTourDriveFolder(ss, params);
      } else if (action === 'renameVehicle') {
        result = renameVehicle(ss, params);
      } else if (action === 'saveGeminiApiKey') {
        result = saveGeminiApiKey(ss, params);
      } else if (action === 'addCustomStatus') {
        result = addCustomStatus(ss, params);
      } else if (action === 'deleteCustomStatus') {
        result = deleteCustomStatus(ss, params);
      } else {
        result = { success: false, message: 'Hành động không hợp lệ!' };
      }
      SpreadsheetApp.flush();
      return jsonResponse(result);
    } finally {
      lock.releaseLock();
    }
  } catch (err) {
    return jsonResponse({ success: false, message: 'Lỗi máy chủ: ' + err.toString() });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateTourFolder(programName, programDate) {
  let rootFolder;
  try {
    rootFolder = DriveApp.getFolderById(ROOT_DRIVE_FOLDER_ID);
  } catch (e) {
    rootFolder = DriveApp.getRootFolder();
  }

  let dateStr = 'NgayChuaRo';
  if (programDate) {
    const d = new Date(programDate);
    if (!isNaN(d.getTime())) {
      const yyyy = d.getFullYear();
      const mm = String(d.getMonth() + 1).padStart(2, '0');
      const dd = String(d.getDate()).padStart(2, '0');
      dateStr = `${yyyy}.${mm}.${dd}`;
    } else {
      dateStr = String(programDate).replace(/[^\w.-]/g, '.');
    }
  }

  const cleanProgName = String(programName || 'Tour').replace(/[\\/:*?"<>|]/g, '_').trim();
  const folderName = `${dateStr}_${cleanProgName}`;

  const folders = rootFolder.getFoldersByName(folderName);
  let tourFolder;
  if (folders.hasNext()) {
    tourFolder = folders.next();
  } else {
    tourFolder = rootFolder.createFolder(folderName);
    try {
      tourFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (err) {
      console.warn('Set sharing error:', err);
    }
  }

  return {
    id: tourFolder.getId(),
    name: folderName,
    url: tourFolder.getUrl()
  };
}

function getInitData(ss, params) {
  const progSheet = ss.getSheetByName(SHEET_PROGRAMS);
  const rows = progSheet.getDataRange().getValues();
  const isAdmin = String(params.pin || '') === ADMIN_PIN;
  const programs = [];
  const colMap = getProgramSheetHeadersMap(progSheet);

  for (let i = 1; i < rows.length; i++) {
    const r = rows[i];
    if (!r[0]) continue;
    const prog = {
      id: String(r[0]),
      name: String(r[1]),
      date: r[2] ? Utilities.formatDate(new Date(r[2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '',
      status: String(r[3] || 'Sắp diễn ra'),
      notice: String(r[4] || ''),
      createdAt: r[5] || '',
      coordinator: String(r[colMap.colCoord] || ''),
      coordinatorPhone: String(r[colMap.colCoordPhone] || ''),
      logistics: String(r[colMap.colLogistics] || ''),
      logisticsPhone: String(r[colMap.colLogisticsPhone] || ''),
      driveFolderId: String(r[colMap.colDriveId] || ''),
      driveFolderLink: String(r[colMap.colDriveLink] || ''),
      endDate: r[colMap.colEndDate] ? Utilities.formatDate(new Date(r[colMap.colEndDate]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : (r[2] ? Utilities.formatDate(new Date(r[2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '')
    };
    if (isAdmin || prog.status === 'Đang diễn ra') {
      programs.push(prog);
    }
  }

  // Lấy danh sách trạng thái xe
  const statusSheet = ss.getSheetByName(SHEET_STATUSES);
  const statusRows = statusSheet.getDataRange().getValues();
  const vehicleStatuses = [];
  for (let j = 1; j < statusRows.length; j++) {
    if (statusRows[j][0]) {
      vehicleStatuses.push({
        name: String(statusRows[j][0]),
        color: String(statusRows[j][1] || '#10b981'),
        isMoving: Boolean(statusRows[j][2])
      });
    }
  }

  return { success: true, programs: programs, statuses: vehicleStatuses };
}

function getVehicleData(ss, params) {
  const progId = String(params.programId || '').trim();
  const vehicleName = String(params.vehicleName || '').trim();
  if (!progId || !vehicleName) return { success: false, message: 'Thiếu thông tin' };

  const dataSheet = ss.getSheetByName(SHEET_DATA);
  const rows = dataSheet.getDataRange().getValues();
  const targetKey = normalizeVehicleKey(vehicleName);

  for (let i = rows.length - 1; i >= 1; i--) {
    const r = rows[i];
    if (String(r[1]) === progId && normalizeVehicleKey(r[3]) === targetKey) {
      let classes = [];
      try { classes = JSON.parse(r[19] || '[]'); } catch (e) { classes = []; }
      
      // Tìm xem xe này đã gửi ảnh điểm danh chưa
      const attRecord = getAttendanceRecord(ss, { programId: progId, vehicleName: vehicleName });

      return {
        success: true,
        found: true,
        data: {
          updatedAt: r[0],
          programId: String(r[1]),
          programName: String(r[2]),
          vehicleName: String(r[3]),
          status: String(r[4] || '🚩 Đang xuất phát'),
          totalStudents: Number(r[5] || 0),
          classSummary: String(r[6] || ''),
          teacherCount: Number(r[7] || 0),
          parentCount: Number(r[8] || 0),
          childCount: Number(r[9] || 0),
          totalPeople: Number(r[10] || 0),
          hdvName: String(r[11] || ''),
          hdvPhone: String(r[12] || '').replace(/^'+/, ''),
          leaderName: String(r[13] || ''),
          leaderPhone: String(r[14] || '').replace(/^'+/, ''),
          vipRoles: String(r[15] || ''),
          driverName: String(r[16] || ''),
          driverPhone: String(r[17] || '').replace(/^'+/, ''),
          licensePlate: String(r[18] || ''),
          classes: classes,
          lat: String(r[20] || '').replace(/^'+/, ''),
          lng: String(r[21] || '').replace(/^'+/, ''),
          gpsTime: r[22] || '',
          attendance: attRecord.found ? attRecord.record : null
        }
      };
    }
  }

  // Nếu chưa có báo cáo chi tiết xe, kiểm tra xem có ảnh điểm danh chưa
  const attRecord = getAttendanceRecord(ss, { programId: progId, vehicleName: vehicleName });
  return {
    success: true,
    found: false,
    attendance: attRecord.found ? attRecord.record : null
  };
}

// Báo cáo tổng hợp toàn đoàn kèm tọa độ GPS
function getSummaryData(ss, params) {
  const progId = String(params.programId || '').trim();
  const dataSheet = ss.getSheetByName(SHEET_DATA);
  const rows = dataSheet.getDataRange().getValues();

  // Load attendance map to quickly attach attendance stats to summary
  const attSheet = ss.getSheetByName(SHEET_ATTENDANCE);
  const attRows = attSheet.getDataRange().getValues();
  const attMap = {};
  for (let k = 1; k < attRows.length; k++) {
    const ar = attRows[k];
    if (ar[1] && (!progId || String(ar[1]) === progId)) {
      const key = String(ar[1]) + '__' + normalizeVehicleKey(ar[2]);
      attMap[key] = {
        time: ar[0],
        fileIds: String(ar[3] || '').split(',').filter(Boolean),
        imageUrls: String(ar[4] || '').split(',').filter(Boolean),
        presentCount: Number(ar[5] || 0),
        absentCount: Number(ar[6] || 0)
      };
    }
  }

  const latestMap = {};
  for (let i = 1; i < rows.length; i++) {
    const r = rows[i];
    if (!r[1] || (progId && String(r[1]) !== progId)) continue;
    const vKey = String(r[1]) + '__' + normalizeVehicleKey(r[3]);
    latestMap[vKey] = {
      updatedAt: r[0],
      programId: String(r[1]),
      programName: String(r[2]),
      vehicleName: String(r[3]),
      status: String(r[4] || '🚩 Đang xuất phát'),
      totalStudents: Number(r[5] || 0),
      classSummary: String(r[6] || ''),
      teacherCount: Number(r[7] || 0),
      parentCount: Number(r[8] || 0),
      childCount: Number(r[9] || 0),
      totalPeople: Number(r[10] || 0),
      hdvName: String(r[11] || ''),
      hdvPhone: String(r[12] || '').replace(/^'+/, ''),
      leaderName: String(r[13] || ''),
      leaderPhone: String(r[14] || '').replace(/^'+/, ''),
      vipRoles: String(r[15] || ''),
      driverName: String(r[16] || ''),
      driverPhone: String(r[17] || '').replace(/^'+/, ''),
      licensePlate: String(r[18] || ''),
      lat: String(r[20] || '').replace(/^'+/, ''),
      lng: String(r[21] || '').replace(/^'+/, ''),
      gpsTime: r[22] || '',
      attendance: attMap[vKey] || null
    };
  }

  let tourInfo = null;
  if (progId) {
    const progSheet = ss.getSheetByName(SHEET_PROGRAMS);
    if (progSheet) {
      const progRows = progSheet.getDataRange().getValues();
      const colMap = getProgramSheetHeadersMap(progSheet);
      for (let p = 1; p < progRows.length; p++) {
        if (String(progRows[p][0]) === progId) {
          tourInfo = {
            id: String(progRows[p][0]),
            name: String(progRows[p][1]),
            date: progRows[p][2] ? Utilities.formatDate(new Date(progRows[p][2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '',
            status: String(progRows[p][3] || ''),
            coordinator: String(progRows[p][colMap.colCoord] || ''),
            coordinatorPhone: String(progRows[p][colMap.colCoordPhone] || ''),
            logistics: String(progRows[p][colMap.colLogistics] || ''),
            logisticsPhone: String(progRows[p][colMap.colLogisticsPhone] || ''),
            driveFolderLink: String(progRows[p][colMap.colDriveLink] || ''),
            endDate: progRows[p][colMap.colEndDate] ? Utilities.formatDate(new Date(progRows[p][colMap.colEndDate]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : (progRows[p][2] ? Utilities.formatDate(new Date(progRows[p][2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '')
          };
          break;
        }
      }
    }
  }

  return {
    success: true,
    vehicles: Object.keys(latestMap).map(k => latestMap[k]),
    tourInfo: tourInfo
  };
}

function saveVehicleData(ss, params) {
  const d = params.data || {};
  const progId = String(d.programId || '').trim();
  const vehicleName = String(d.vehicleName || '').trim();
  if (!progId || !vehicleName) return { success: false, message: 'Vui lòng chọn tour và số xe!' };

  const dataSheet = ss.getSheetByName(SHEET_DATA);
  const rows = dataSheet.getDataRange().getValues();
  const targetKey = normalizeVehicleKey(vehicleName);
  const nowStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');

  const latStr = d.lat ? ("'" + String(d.lat).replace(/^'+/, '')) : '';
  const lngStr = d.lng ? ("'" + String(d.lng).replace(/^'+/, '')) : '';
  const gpsTimeStr = d.gpsTime || (d.lat ? nowStr : '');

  const rowValues = [
    nowStr,
    progId,
    d.programName || '',
    vehicleName,
    d.status || '🚩 Đang xuất phát',
    Number(d.totalStudents || 0),
    d.classSummary || '',
    Number(d.teacherCount || 0),
    Number(d.parentCount || 0),
    Number(d.childCount || 0),
    Number(d.totalPeople || 0),
    d.hdvName || '',
    "'" + String(d.hdvPhone || '').replace(/^'+/, ''),
    d.leaderName || '',
    "'" + String(d.leaderPhone || '').replace(/^'+/, ''),
    d.vipRoles || '',
    d.driverName || '',
    "'" + String(d.driverPhone || '').replace(/^'+/, ''),
    d.licensePlate || '',
    JSON.stringify(d.classes || []),
    latStr,
    lngStr,
    gpsTimeStr
  ];

  let targetRowIndex = -1;
  for (let i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][1]) === progId && normalizeVehicleKey(rows[i][3]) === targetKey) {
      targetRowIndex = i + 1;
      break;
    }
  }

  if (targetRowIndex !== -1) {
    dataSheet.getRange(targetRowIndex, 1, 1, rowValues.length).setValues([rowValues]);
  } else {
    dataSheet.appendRow(rowValues);
  }

  // Tự động lưu thông tin điểm danh kèm theo nếu có
  if (d.attendance && d.attendance.students && d.attendance.students.length > 0) {
    try {
      saveAttendanceRecordInSheet(
        ss,
        progId,
        vehicleName,
        d.attendance.fileIds || [],
        d.attendance.imageUrls || [],
        d.attendance.presentCount || 0,
        d.attendance.absentCount || 0,
        d.attendance.students
      );
    } catch (e) {
      console.warn('Lỗi lưu điểm danh kèm xe:', e);
    }
  }

  return { success: true, message: 'Đã lưu & đồng bộ vị trí GPS cho ' + vehicleName, updatedAt: nowStr };
}

function getTourDetails(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const progId = String(params.programId || '').trim();
  const progSheet = ss.getSheetByName(SHEET_PROGRAMS);
  const rows = progSheet.getDataRange().getValues();
  const colMap = getProgramSheetHeadersMap(progSheet);

  let tour = null;
  let rowIndex = -1;
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === progId) {
      rowIndex = i + 1;
      tour = {
        id: String(rows[i][0]),
        name: String(rows[i][1]),
        date: rows[i][2] ? Utilities.formatDate(new Date(rows[i][2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '',
        status: String(rows[i][3] || 'Đang diễn ra'),
        notice: String(rows[i][4] || ''),
        coordinator: String(rows[i][colMap.colCoord] || ''),
        coordinatorPhone: String(rows[i][colMap.colCoordPhone] || ''),
        logistics: String(rows[i][colMap.colLogistics] || ''),
        logisticsPhone: String(rows[i][colMap.colLogisticsPhone] || ''),
        driveFolderId: String(rows[i][colMap.colDriveId] || ''),
        driveFolderLink: String(rows[i][colMap.colDriveLink] || ''),
        endDate: rows[i][colMap.colEndDate] ? Utilities.formatDate(new Date(rows[i][colMap.colEndDate]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : (rows[i][2] ? Utilities.formatDate(new Date(rows[i][2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '')
      };
      break;
    }
  }

  if (!tour) return { success: false, message: 'Không tìm thấy tour!' };

  // Tự động đảm bảo đã tạo Folder Google Drive cho Tour
  if (!tour.driveFolderId || !tour.driveFolderLink) {
    try {
      const folderInfo = getOrCreateTourFolder(tour.name, tour.date);
      tour.driveFolderId = folderInfo.id;
      tour.driveFolderLink = folderInfo.url;
      progSheet.getRange(rowIndex, colMap.colDriveId + 1).setValue(folderInfo.id);
      progSheet.getRange(rowIndex, colMap.colDriveLink + 1).setValue(folderInfo.url);
    } catch (err) {
      console.warn('Lỗi tạo folder Drive:', err);
    }
  }

  // Lấy danh sách lớp và học sinh đã tải lên
  const studentSheet = ss.getSheetByName(SHEET_STUDENT_LISTS);
  const stuRows = studentSheet.getDataRange().getValues();
  const classSummaryMap = {};

  for (let j = 1; j < stuRows.length; j++) {
    if (String(stuRows[j][0]) === progId) {
      const cName = String(stuRows[j][1]);
      if (!classSummaryMap[cName]) {
        classSummaryMap[cName] = { className: cName, count: 0, sampleStudents: [] };
      }
      classSummaryMap[cName].count++;
      if (classSummaryMap[cName].sampleStudents.length < 5) {
        classSummaryMap[cName].sampleStudents.push(String(stuRows[j][3]));
      }
    }
  }

  return {
    success: true,
    tour: tour,
    classes: Object.keys(classSummaryMap).map(k => classSummaryMap[k])
  };
}

function updateTourDetails(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const progId = String(params.programId || '').trim();
  const progSheet = ss.getSheetByName(SHEET_PROGRAMS);
  const rows = progSheet.getDataRange().getValues();
  const colMap = getProgramSheetHeadersMap(progSheet);

  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === progId) {
      const rowIdx = i + 1;
      if (params.name) progSheet.getRange(rowIdx, 2).setValue(params.name);
      if (params.date) progSheet.getRange(rowIdx, 3).setValue(params.date);
      if (params.coordinator !== undefined) progSheet.getRange(rowIdx, colMap.colCoord + 1).setValue(params.coordinator);
      if (params.coordinatorPhone !== undefined) progSheet.getRange(rowIdx, colMap.colCoordPhone + 1).setValue(params.coordinatorPhone);
      if (params.logistics !== undefined) progSheet.getRange(rowIdx, colMap.colLogistics + 1).setValue(params.logistics);
      if (params.logisticsPhone !== undefined) progSheet.getRange(rowIdx, colMap.colLogisticsPhone + 1).setValue(params.logisticsPhone);
      if (params.endDate !== undefined) {
        progSheet.getRange(rowIdx, colMap.colEndDate + 1).setValue(params.endDate || params.date || '');
      } else if (params.date) {
        const curEnd = progSheet.getRange(rowIdx, colMap.colEndDate + 1).getValue();
        if (!curEnd) progSheet.getRange(rowIdx, colMap.colEndDate + 1).setValue(params.date);
      }

      // Cập nhật hoặc đảm bảo folder Google Drive
      const curProgName = params.name || rows[i][1];
      const curProgDate = params.date || rows[i][2];
      try {
        const folderInfo = getOrCreateTourFolder(curProgName, curProgDate);
        progSheet.getRange(rowIdx, colMap.colDriveId + 1).setValue(folderInfo.id);
        progSheet.getRange(rowIdx, colMap.colDriveLink + 1).setValue(folderInfo.url);
      } catch (e) {}

      return { success: true, message: 'Đã cập nhật thông tin quản trị tour thành công!' };
    }
  }
  return { success: false, message: 'Không tìm thấy tour để cập nhật!' };
}

// Lưu danh sách học sinh theo lớp trích xuất từ file Excel
function saveTourClassLists(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const progId = String(params.programId || '').trim();
  const classes = params.classes || [];
  if (!progId || classes.length === 0) {
    return { success: false, message: 'Không có dữ liệu lớp hoặc học sinh để lưu!' };
  }

  // Lưu file Excel gốc lên Google Drive của tour nếu có
  if (params.excelFiles && params.excelFiles.length > 0) {
    try {
      const tourDetails = getTourDetails(ss, { pin: ADMIN_PIN, programId: progId });
      const folderId = tourDetails.tour && tourDetails.tour.driveFolderId;
      if (folderId) {
        const folder = DriveApp.getFolderById(folderId);
        params.excelFiles.forEach(f => {
          if (f.base64 && f.name) {
            const cleanBase = f.base64.replace(/^data:[^;]+;base64,/, '');
            const blob = Utilities.newBlob(Utilities.base64Decode(cleanBase), 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', f.name);
            const savedFile = folder.createFile(blob);
            try { savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
          }
        });
      }
    } catch (e) {
      console.warn('Lưu file Excel lên Drive thất bại:', e);
    }
  }

  const studentSheet = ss.getSheetByName(SHEET_STUDENT_LISTS);
  const rows = studentSheet.getDataRange().getValues();

  // Xác định các lớp cần ghi đè
  const targetClassKeys = new Set(classes.map(c => normalizeVehicleKey(c.className)));

  // Xóa danh sách cũ của các lớp tương ứng trong chương trình này (từ dưới lên trên)
  for (let i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][0]) === progId && targetClassKeys.has(normalizeVehicleKey(rows[i][1]))) {
      studentSheet.deleteRow(i + 1);
    }
  }

  // Thêm dữ liệu học sinh mới
  let totalSaved = 0;
  classes.forEach(c => {
    const cName = String(c.className || '').trim();
    const stuList = c.students || [];
    stuList.forEach(s => {
      const stuName = String(s.name || '').trim();
      const cleanName = removeVietnameseTones(stuName).toLowerCase();
      studentSheet.appendRow([
        progId,
        cName,
        s.stt || (totalSaved + 1),
        stuName,
        cleanName,
        ''
      ]);
      totalSaved++;
    });
  });

  return {
    success: true,
    message: `Đã nạp thành công ${classes.length} lớp (${totalSaved} học sinh)!`,
    totalClasses: classes.length,
    totalStudents: totalSaved
  };
}

// Lấy danh sách học sinh gốc của lớp / xe để đối chiếu
function getClassStudentList(ss, params) {
  const progId = String(params.programId || '').trim();
  const vehicleOrClassName = String(params.className || params.vehicleName || '').trim();
  if (!progId || !vehicleOrClassName) return { success: false, students: [] };

  const studentSheet = ss.getSheetByName(SHEET_STUDENT_LISTS);
  const rows = studentSheet.getDataRange().getValues();
  const targetKey = normalizeVehicleKey(vehicleOrClassName);

  const students = [];
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === progId && normalizeVehicleKey(rows[i][1]) === targetKey) {
      students.push({
        stt: Number(rows[i][2] || (students.length + 1)),
        name: String(rows[i][3]),
        cleanName: String(rows[i][4])
      });
    }
  }

  return { success: true, students: students };
}

// 6. CHỤP ẢNH, LƯU GOOGLE DRIVE & GEMINI API PHÂN TÍCH ĐIỂM DANH
// ==============================================================
function uploadAndAnalyzeAttendanceImage(ss, params) {
  const progId = String(params.programId || '').trim();
  const progName = String(params.programName || '').trim();
  const tourDate = String(params.tourDate || '').trim();
  const vehicleName = String(params.vehicleName || '').trim();
  const images = params.images || []; // Mảng 1 hoặc 2 base64 strings

  if (!progId || !vehicleName) {
    return { success: false, message: 'Thiếu thông tin chương trình hoặc số xe!' };
  }
  if (images.length === 0) {
    return { success: false, message: 'Vui lòng cung cấp ít nhất 1 ảnh danh sách!' };
  }
  if (images.length > 2) {
    return { success: false, message: 'Mỗi lượt gửi chỉ được tối đa 2 ảnh!' };
  }

  // 1. Lấy hoặc tạo Folder Google Drive cho Tour
  let tourFolder = null;
  let driveWarning = '';
  try {
    const tourFolderInfo = getOrCreateTourFolder(progName, tourDate);
    if (tourFolderInfo && tourFolderInfo.id) {
      tourFolder = DriveApp.getFolderById(tourFolderInfo.id);
    }
  } catch (err) {
    console.warn('Cảnh báo kết nối Google Drive:', err);
    driveWarning = 'Chưa lưu ảnh lên Google Drive do Apps Script chưa được cấp quyền DriveApp (Hãy mở Apps Script chọn hàm capQuyenVaKiemTraGoogleDrive và ấn Chạy để cấp quyền 1 lần duy nhất).';
  }

  // 2. Định dạng tên file: [tên xe/lớp]_[Tên chương trình/tên trường]_[Ngày thực hiện]
  let datePart = tourDate ? tourDate.replace(/[^\w.-]/g, '.') : Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy.MM.dd');
  const cleanVehicle = vehicleName.replace(/[\\/:*?"<>|]/g, '_').trim();
  const cleanProg = progName.replace(/[\\/:*?"<>|]/g, '_').trim();
  const baseFileName = `${cleanVehicle}_${cleanProg}_${datePart}`;

  // 3. Lưu ảnh vào Google Drive nếu thư mục sẵn sàng & Cấp quyền xem công khai
  const fileIds = [];
  const imageUrls = [];

  if (tourFolder) {
    for (let idx = 0; idx < images.length; idx++) {
      try {
        const rawBase64 = images[idx];
        const cleanBase64 = rawBase64.replace(/^data:[^;]+;base64,/, '');
        const fileName = images.length > 1 ? `${baseFileName}_${idx + 1}.jpg` : `${baseFileName}.jpg`;

        const blob = Utilities.newBlob(Utilities.base64Decode(cleanBase64), 'image/jpeg', fileName);
        const savedFile = tourFolder.createFile(blob);
        try {
          savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        } catch (e) {
          console.warn('Set sharing image error:', e);
        }

        const fid = savedFile.getId();
        fileIds.push(fid);
        const directUrl = 'https://drive.google.com/thumbnail?id=' + fid + '&sz=w1600';
        imageUrls.push(directUrl);
      } catch (fErr) {
        console.warn('Lỗi lưu ảnh lên Drive:', fErr);
      }
    }
  }

  // 4. Phân tích ảnh bằng Gemini API
  let geminiApiKey = params.geminiApiKey || PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY') || DEFAULT_GEMINI_API_KEY;
  let ocrStudents = [];
  try {
    ocrStudents = callGeminiForAttendance(images, geminiApiKey);
  } catch (ocrErr) {
    console.warn('Gemini OCR lỗi:', ocrErr);
    saveAttendanceRecordInSheet(ss, progId, vehicleName, fileIds, imageUrls, 0, 0, []);
    return {
      success: false,
      message: 'Gemini AI phân tích ảnh thất bại: ' + ocrErr.message,
      warning: (driveWarning ? driveWarning + ' | ' : '') + ocrErr.message,
      imageUrls: imageUrls,
      fileIds: fileIds,
      students: [],
      totalStudents: 0,
      presentCount: 0,
      absentCount: 0
    };
  }

  if (!ocrStudents || !Array.isArray(ocrStudents) || ocrStudents.length === 0) {
    saveAttendanceRecordInSheet(ss, progId, vehicleName, fileIds, imageUrls, 0, 0, []);
    return {
      success: false,
      message: 'Gemini AI không tìm thấy danh sách học sinh nào trong ảnh chụp. Vui lòng chụp rõ nét hơn!',
      imageUrls: imageUrls,
      fileIds: fileIds,
      students: [],
      totalStudents: 0,
      presentCount: 0,
      absentCount: 0
    };
  }

  // 5. Đối chiếu kết quả OCR với danh sách học sinh gốc từ Excel
  const officialRes = getClassStudentList(ss, { programId: progId, vehicleName: vehicleName });
  const officialStudents = officialRes.students || [];

  const matched = matchAttendanceWithOfficialList(officialStudents, ocrStudents);

  // 6. Lưu kết quả điểm danh vào Sheet DiemDanhAnh (Ghi đè nếu đã tồn tại)
  saveAttendanceRecordInSheet(
    ss,
    progId,
    vehicleName,
    fileIds,
    imageUrls,
    matched.presentCount,
    matched.absentCount,
    matched.students
  );

  // Đồng thời cập nhật tổng sĩ số học sinh cho xe ở Sheet TongHopDuLieu nếu đã có dòng xe
  updateVehicleStudentCountIfFound(ss, progId, vehicleName, matched.students.length);

  return {
    success: true,
    warning: driveWarning || undefined,
    message: `Điểm danh thành công: ${matched.presentCount}/${matched.totalStudents} học sinh có mặt!` + (driveWarning ? ` (${driveWarning})` : ''),
    imageUrls: imageUrls,
    fileIds: fileIds,
    totalStudents: matched.totalStudents,
    presentCount: matched.presentCount,
    absentCount: matched.absentCount,
    students: matched.students
  };
}

// Gọi Gemini API để nhận diện danh sách học sinh và dấu tích điểm danh
function callGeminiForAttendance(imagesBase64, apiKey) {
  if (!apiKey) {
    apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY') || DEFAULT_GEMINI_API_KEY;
  }

  const parts = [
    {
      text: "Bạn là chuyên gia phân tích danh sách học sinh từ ảnh chụp bảng điểm danh lớp học đi dã ngoại.\n" +
            "Nhiệm vụ của bạn: Đọc toàn bộ danh sách học sinh trong tất cả các ảnh và xác định học sinh nào CÓ MẶT, học sinh nào VẮNG MẶT.\n\n" +
            "QUY TẮC NHẬN DIỆN CỰC KỲ QUAN TRỌNG:\n" +
            "1. Hướng dẫn viên điểm danh bằng cách đánh dấu cùng hàng với tên học sinh: có thể là dấu tích 'v', đánh 'x', gạch ngang '-', khoanh tròn hoặc dấu bất kỳ ở trước hoặc sau tên.\n" +
            "2. Bất kỳ học sinh nào có đánh dấu cùng hàng tên -> ĐANG CÓ MẶT (present: true).\n" +
            "3. Học sinh nào KHÔNG CÓ ĐÁNH DẤU gì cùng hàng tên -> MẶC ĐỊNH LÀ VẮNG MẶT (present: false).\n" +
            "4. Đọc chuẩn xác họ và tên tiếng Việt (họ tên đầy đủ có dấu nếu đọc được), và số thứ tự (stt).\n" +
            "5. Nếu có nhiều ảnh, hãy ghép toàn bộ học sinh của các ảnh thành MỘT danh sách duy nhất không trùng lặp theo đúng thứ tự.\n\n" +
            "Hãy trả về kết quả ĐÚNG ĐỊNH DẠNG JSON sau (chỉ trả về JSON thuần, không bọc markdown ```json):\n" +
            "[\n" +
            "  {\"stt\": 1, \"name\": \"Nguyễn Văn An\", \"present\": true},\n" +
            "  {\"stt\": 2, \"name\": \"Trần Thị Bình\", \"present\": false}\n" +
            "]"
    }
  ];

  imagesBase64.forEach(img => {
    let mime = 'image/jpeg';
    let data = img;
    const match = img.match(/^data:([^;]+);base64,(.+)$/);
    if (match) {
      mime = match[1];
      data = match[2];
    }
    parts.push({
      inlineData: {
        mimeType: mime,
        data: data
      }
    });
  });

  const payload = {
    contents: [{ parts: parts }],
    generationConfig: {
      temperature: 0.1,
      responseMimeType: "application/json"
    }
  };

  const candidateModels = [
    'gemini-3.1-flash-lite',
    'gemini-3.5-flash-lite',
    'gemini-3.5-flash',
    'gemini-3.1-flash-image',
    'gemini-flash-latest',
    'gemini-flash-lite-latest',
    'gemini-2.5-flash-lite',
    'gemini-2.5-flash',
    'gemini-1.5-flash-latest'
  ];

  let lastErrorText = '';
  for (let m = 0; m < candidateModels.length; m++) {
    const modelName = candidateModels[m];
    const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + modelName + ':generateContent?key=' + encodeURIComponent(apiKey);
    const response = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    const code = response.getResponseCode();
    const text = response.getContentText();
    if (code === 200) {
      const resJson = JSON.parse(text);
      const rawContent = resJson.candidates && resJson.candidates[0] && resJson.candidates[0].content && resJson.candidates[0].content.parts && resJson.candidates[0].content.parts[0] && resJson.candidates[0].content.parts[0].text;
      if (!rawContent) throw new Error('Gemini không phản hồi dữ liệu');

      let cleanJson = rawContent.trim();
      if (cleanJson.startsWith('```json')) cleanJson = cleanJson.replace(/^```json\s*/i, '').replace(/```\s*$/, '').trim();
      else if (cleanJson.startsWith('```')) cleanJson = cleanJson.replace(/^```\s*/, '').replace(/```\s*$/, '').trim();

      return JSON.parse(cleanJson);
    } else if (code === 404) {
      lastErrorText = text;
      continue; // Tự động thử model tiếp theo
    } else {
      throw new Error('Lỗi từ Gemini API (Mã ' + code + '): ' + text);
    }
  }

  // Tự động khám phá qua ModelService.ListModels nếu tất cả candidate đều 404
  try {
    const listUrl = 'https://generativelanguage.googleapis.com/v1beta/models?key=' + encodeURIComponent(apiKey);
    const listResp = UrlFetchApp.fetch(listUrl, { method: 'get', muteHttpExceptions: true });
    if (listResp.getResponseCode() === 200) {
      const listData = JSON.parse(listResp.getContentText());
      if (listData.models && Array.isArray(listData.models)) {
        for (let i = 0; i < listData.models.length; i++) {
          const mObj = listData.models[i];
          if (mObj.supportedGenerationMethods && mObj.supportedGenerationMethods.indexOf('generateContent') !== -1) {
            const mName = mObj.name.replace(/^models\//, '');
            if (candidateModels.indexOf(mName) !== -1) continue;
            const dynUrl = 'https://generativelanguage.googleapis.com/v1beta/models/' + mName + ':generateContent?key=' + encodeURIComponent(apiKey);
            const dynResp = UrlFetchApp.fetch(dynUrl, {
              method: 'post',
              contentType: 'application/json',
              payload: JSON.stringify(payload),
              muteHttpExceptions: true
            });
            if (dynResp.getResponseCode() === 200) {
              const resJson = JSON.parse(dynResp.getContentText());
              const rawContent = resJson.candidates && resJson.candidates[0] && resJson.candidates[0].content && resJson.candidates[0].content.parts && resJson.candidates[0].content.parts[0] && resJson.candidates[0].content.parts[0].text;
              if (rawContent) {
                let cleanJson = rawContent.trim();
                if (cleanJson.startsWith('```json')) cleanJson = cleanJson.replace(/^```json\s*/i, '').replace(/```\s*$/, '').trim();
                else if (cleanJson.startsWith('```')) cleanJson = cleanJson.replace(/^```\s*/, '').replace(/```\s*$/, '').trim();
                return JSON.parse(cleanJson);
              }
            }
          }
        }
      }
    }
  } catch (e) {
    console.warn('Lỗi ListModels backend:', e);
  }

  throw new Error('Không tìm thấy model Gemini phù hợp: ' + lastErrorText);
}

// Thuật toán đối chiếu thông minh: So sánh không dấu, xử lý trùng tên
function matchAttendanceWithOfficialList(officialStudents, ocrStudents) {
  function cleanStr(s) {
    return removeVietnameseTones(s).toLowerCase().replace(/[^a-z0-9]/g, ' ').replace(/\s+/g, ' ').trim();
  }

  // Nếu chưa có danh sách chính thức từ Excel, dùng trực tiếp danh sách OCR của Gemini
  if (!officialStudents || officialStudents.length === 0) {
    let pCount = 0, aCount = 0;
    const directList = (ocrStudents || []).map((o, idx) => {
      const isPres = Boolean(o.present);
      if (isPres) pCount++; else aCount++;
      return {
        stt: o.stt !== undefined ? Number(o.stt) : (idx + 1),
        name: o.name || ('Học sinh ' + (idx + 1)),
        cleanName: cleanStr(o.name || ''),
        present: isPres
      };
    });
    return {
      students: directList,
      totalStudents: directList.length,
      presentCount: pCount,
      absentCount: aCount
    };
  }

  // Xử lý danh sách OCR
  const ocrList = (ocrStudents || []).map((o, idx) => ({
    rawName: o.name || '',
    cleanName: cleanStr(o.name || ''),
    present: Boolean(o.present),
    stt: o.stt !== undefined ? Number(o.stt) : (idx + 1),
    used: false
  }));

  // Đếm tần suất tên không dấu trong danh sách chính thức để nhận diện học sinh trùng tên
  const officialNameCount = {};
  officialStudents.forEach(s => {
    const c = cleanStr(s.name);
    officialNameCount[c] = (officialNameCount[c] || 0) + 1;
  });

  const result = [];
  let presentCount = 0;
  let absentCount = 0;

  officialStudents.forEach(stu => {
    const cName = cleanStr(stu.name);
    const stuStt = Number(stu.stt);
    const isDuplicate = (officialNameCount[cName] > 1);

    let matchedOcr = null;

    // Chiến lược 1: Nếu trùng tên, ưu tiên khớp theo cả STT gần nhất
    if (isDuplicate) {
      matchedOcr = ocrList.find(o => !o.used && o.cleanName === cName && Math.abs(o.stt - stuStt) <= 1);
    }

    // Chiến lược 2: Khớp chính xác tên không dấu chưa dùng
    if (!matchedOcr) {
      matchedOcr = ocrList.find(o => !o.used && o.cleanName === cName);
    }

    // Chiến lược 3: Tìm kiếm theo từ khóa họ tên tương tự (> 80% khớp ký tự)
    if (!matchedOcr) {
      let maxSim = 0;
      let candidate = null;
      ocrList.filter(o => !o.used).forEach(o => {
        const sim = calcTextSimilarity(cName, o.cleanName);
        if (sim >= 0.8 && sim > maxSim) {
          maxSim = sim;
          candidate = o;
        }
      });
      if (candidate) matchedOcr = candidate;
    }

    let isPresent = false;
    let ocrFoundName = '';
    if (matchedOcr) {
      matchedOcr.used = true;
      isPresent = Boolean(matchedOcr.present);
      ocrFoundName = matchedOcr.rawName;
    } else {
      // Học sinh không đánh dấu mặc định là vắng
      isPresent = false;
    }

    if (isPresent) presentCount++;
    else absentCount++;

    result.push({
      stt: stu.stt,
      name: stu.name, // Giữ nguyên tên chuẩn có dấu từ Excel
      cleanName: cName,
      present: isPresent,
      ocrName: ocrFoundName
    });
  });

  return {
    students: result,
    totalStudents: result.length,
    presentCount: presentCount,
    absentCount: absentCount
  };
}

function calcTextSimilarity(s1, s2) {
  if (s1 === s2) return 1.0;
  if (!s1 || !s2) return 0.0;
  const longer = s1.length > s2.length ? s1 : s2;
  const shorter = s1.length > s2.length ? s2 : s1;
  if (longer.includes(shorter)) return shorter.length / longer.length;
  
  // Bigram Dice similarity
  const getBigrams = str => {
    const s = new Set();
    for (let i = 0; i < str.length - 1; i++) s.add(str.substr(i, 2));
    return s;
  };
  const b1 = getBigrams(s1);
  const b2 = getBigrams(s2);
  let intersection = 0;
  b1.forEach(bg => { if (b2.has(bg)) intersection++; });
  return (2.0 * intersection) / (b1.size + b2.size || 1);
}

// Lưu hoặc cập nhật bản ghi điểm danh
function saveAttendanceRecordInSheet(ss, progId, vehicleName, fileIds, imageUrls, presentCount, absentCount, students) {
  let attSheet = ss.getSheetByName(SHEET_ATTENDANCE);
  if (!attSheet) {
    initSheets();
    attSheet = ss.getSheetByName(SHEET_ATTENDANCE);
  }
  if (!attSheet) return;

  const rows = attSheet.getDataRange().getValues();
  const targetKey = normalizeVehicleKey(vehicleName);
  const nowStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');

  const fileIdsStr = (fileIds || []).join(',');
  // Lọc bỏ base64 ảnh thô quá 500 ký tự để không làm tràn giới hạn 50.000 ký tự 1 ô bảng tính Google Sheet
  const cleanUrls = (imageUrls || []).map(function(u) {
    const s = String(u);
    return (s.indexOf('data:') === 0 && s.length > 500) ? '' : s;
  }).filter(Boolean);
  const imageUrlsStr = cleanUrls.join(',');
  const studentsJson = JSON.stringify(students || []);

  let foundRowIdx = -1;
  for (let i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][1]) === progId && normalizeVehicleKey(rows[i][2]) === targetKey) {
      foundRowIdx = i + 1;
      break;
    }
  }

  const rowValues = [
    nowStr,
    progId,
    vehicleName,
    fileIdsStr,
    imageUrlsStr,
    Number(presentCount || 0),
    Number(absentCount || 0),
    studentsJson
  ];

  if (foundRowIdx !== -1) {
    attSheet.getRange(foundRowIdx, 1, 1, rowValues.length).setValues([rowValues]);
  } else {
    attSheet.appendRow(rowValues);
  }
}

function handleSaveAttendanceRecord(ss, params) {
  const progId = String(params.programId || '').trim();
  const vehicleName = String(params.vehicleName || '').trim();
  if (!progId || !vehicleName) return { success: false, message: 'Thiếu thông tin chương trình hoặc xe' };

  saveAttendanceRecordInSheet(
    ss,
    progId,
    vehicleName,
    params.fileIds || [],
    params.imageUrls || [],
    params.presentCount || 0,
    params.absentCount || 0,
    params.students || []
  );

  updateVehicleStudentCountIfFound(ss, progId, vehicleName, (params.students || []).length);
  return { success: true, message: 'Đã lưu bản ghi điểm danh vào Google Sheet thành công!' };
}

function handleCreateTourDriveFolder(ss, params) {
  const progName = String(params.programName || '').trim();
  const progDate = String(params.tourDate || '').trim();
  const progId = String(params.programId || '').trim();
  try {
    const folderInfo = getOrCreateTourFolder(progName, progDate);
    if (progId) {
      const progSheet = ss.getSheetByName(SHEET_PROGRAMS);
      const rows = progSheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === progId) {
          progSheet.getRange(i + 1, 9).setValue(folderInfo.id);
          progSheet.getRange(i + 1, 10).setValue(folderInfo.url);
          break;
        }
      }
    }
    return { success: true, folderId: folderInfo.id, folderName: folderInfo.name, folderUrl: folderInfo.url };
  } catch (err) {
    return { success: false, message: 'Lỗi tạo thư mục Google Drive: ' + err.toString() };
  }
}

function getAttendanceRecord(ss, params) {
  const progId = String(params.programId || '').trim();
  const vehicleName = String(params.vehicleName || params.className || '').trim();
  if (!progId || !vehicleName) return { success: false, found: false };

  const attSheet = ss.getSheetByName(SHEET_ATTENDANCE);
  if (!attSheet) return { success: true, found: false };
  const rows = attSheet.getDataRange().getValues();
  const targetKey = normalizeVehicleKey(vehicleName);

  for (let i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][1]) === progId && normalizeVehicleKey(rows[i][2]) === targetKey) {
      let stuList = [];
      try { stuList = JSON.parse(rows[i][7] || '[]'); } catch (e) { stuList = []; }
      return {
        success: true,
        found: true,
        record: {
          updatedAt: rows[i][0],
          programId: String(rows[i][1]),
          vehicleName: String(rows[i][2]),
          fileIds: String(rows[i][3] || '').split(',').filter(Boolean),
          imageUrls: String(rows[i][4] || '').split(',').filter(Boolean),
          presentCount: Number(rows[i][5] || 0),
          absentCount: Number(rows[i][6] || 0),
          students: stuList
        }
      };
    }
  }
  return { success: true, found: false };
}

function updateVehicleStudentCountIfFound(ss, progId, vehicleName, totalStudents) {
  try {
    const dataSheet = ss.getSheetByName(SHEET_DATA);
    const rows = dataSheet.getDataRange().getValues();
    const targetKey = normalizeVehicleKey(vehicleName);
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][1]) === progId && normalizeVehicleKey(rows[i][3]) === targetKey) {
        dataSheet.getRange(i + 1, 6).setValue(totalStudents);
        break;
      }
    }
  } catch (e) {}
}

function renameVehicle(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const progId = String(params.programId || '').trim();
  const oldName = String(params.oldVehicleName || '').trim();
  const newName = String(params.newVehicleName || '').trim();

  if (!progId || !oldName || !newName) {
    return { success: false, message: 'Vui lòng cung cấp đầy đủ tên xe cũ và mới!' };
  }

  const oldKey = normalizeVehicleKey(oldName);
  let updatedAny = false;

  // 1. Cập nhật trong Sheet TongHopDuLieu
  const dataSheet = ss.getSheetByName(SHEET_DATA);
  const dataRows = dataSheet.getDataRange().getValues();
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][1]) === progId && normalizeVehicleKey(dataRows[i][3]) === oldKey) {
      dataSheet.getRange(i + 1, 4).setValue(newName);
      updatedAny = true;
    }
  }

  // 2. Cập nhật trong Sheet DanhSachHocSinhGoc
  const studentSheet = ss.getSheetByName(SHEET_STUDENT_LISTS);
  const stuRows = studentSheet.getDataRange().getValues();
  for (let j = 1; j < stuRows.length; j++) {
    if (String(stuRows[j][0]) === progId && normalizeVehicleKey(stuRows[j][1]) === oldKey) {
      studentSheet.getRange(j + 1, 2).setValue(newName);
      updatedAny = true;
    }
  }

  // 3. Cập nhật trong Sheet DiemDanhAnh
  const attSheet = ss.getSheetByName(SHEET_ATTENDANCE);
  const attRows = attSheet.getDataRange().getValues();
  for (let k = 1; k < attRows.length; k++) {
    if (String(attRows[k][1]) === progId && normalizeVehicleKey(attRows[k][2]) === oldKey) {
      attSheet.getRange(k + 1, 3).setValue(newName);
      updatedAny = true;
    }
  }

  return {
    success: true,
    message: `Đã đổi tên xe từ "${oldName}" thành "${newName}" trên toàn bộ hệ thống!`,
    newVehicleName: newName
  };
}

function saveGeminiApiKey(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const key = String(params.key || '').trim();
  if (!key) return { success: false, message: 'Vui lòng nhập API Key hợp lệ!' };

  PropertiesService.getScriptProperties().setProperty('GEMINI_API_KEY', key);
  return { success: true, message: 'Đã lưu cấu hình Gemini API Key trên máy chủ thành công!' };
}

function getGeminiApiKeyStatus(ss, params) {
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY') || DEFAULT_GEMINI_API_KEY;
  return {
    success: true,
    hasKey: Boolean(key),
    maskedKey: key ? (key.substring(0, 4) + '••••••••' + key.substring(key.length - 4)) : ''
  };
}

function addCustomStatus(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const name = String(params.name || '').trim();
  const color = String(params.color || '#10b981').trim();
  const isMoving = Boolean(params.isMoving);

  if (!name) return { success: false, message: 'Vui lòng nhập tên trạng thái!' };

  const statusSheet = ss.getSheetByName(SHEET_STATUSES);
  statusSheet.appendRow([name, color, isMoving]);
  return { success: true, message: 'Đã thêm trạng thái mới thành công!' };
}

function deleteCustomStatus(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const name = String(params.name || '').trim();
  const statusSheet = ss.getSheetByName(SHEET_STATUSES);
  const rows = statusSheet.getDataRange().getValues();

  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === name) {
      statusSheet.deleteRow(i + 1);
      return { success: true, message: 'Đã xóa trạng thái: ' + name };
    }
  }
  return { success: false, message: 'Không tìm thấy trạng thái để xóa!' };
}

function createProgram(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const progSheet = ss.getSheetByName(SHEET_PROGRAMS);
  const id = 'TOUR_' + new Date().getTime();
  const nowStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  const progName = params.name || 'Tour mới';
  const progDate = params.date || '';

  let folderId = '';
  let folderUrl = '';
  try {
    const fInfo = getOrCreateTourFolder(progName, progDate);
    folderId = fInfo.id;
    folderUrl = fInfo.url;
  } catch (e) {}

  progSheet.appendRow([
    id, progName, progDate, params.status || 'Đang diễn ra', '', nowStr,
    params.coordinator || '', params.logistics || '', folderId, folderUrl
  ]);
  return { success: true, message: 'Đã tạo chương trình thành công!', programId: id };
}

function updateProgramStatus(ss, params) {
  if (String(params.pin || '') !== ADMIN_PIN) return { success: false, message: 'Mã PIN không đúng!' };
  const progSheet = ss.getSheetByName(SHEET_PROGRAMS);

    }
  }
  return { success: false, message: 'Không tìm thấy tour!' };
}
